cmake_minimum_required(VERSION 3.24)
project(mini_risc_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED VROOT)
  message(FATAL_ERROR "Pass -DVROOT=<Verilator package directory>")
endif()
foreach(model smoke mini_risc_cpu mini_risc_soc unit_top)
  if(EXISTS "${CMAKE_SOURCE_DIR}/build/generated/${model}/V${model}.h")
    file(GLOB model_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/build/generated/${model}/*.cpp")
    add_executable(${model}_sim sim/${model}.cpp ${model_sources}
      "${VROOT}/include/verilated.cpp" "${VROOT}/include/verilated_threads.cpp")
    target_include_directories(${model}_sim PRIVATE "${CMAKE_SOURCE_DIR}/build/generated/${model}"
      "${VROOT}/include" "${VROOT}/include/vltstd" sim)
    target_compile_definitions(${model}_sim PRIVATE VM_TRACE=0 VM_COVERAGE=0 VL_TIME_CONTEXT)
    if(MSVC)
      target_compile_options(${model}_sim PRIVATE /bigobj /Zc:__cplusplus /wd4100 /wd4244 /wd4267 /wd4458 /wd4996)
    endif()
  endif()
endforeach()
